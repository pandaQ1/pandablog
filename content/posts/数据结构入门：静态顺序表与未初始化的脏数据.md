```c
#include <stdio.h>//标准输入输出库，包含printf、scanf等函数
#define MAXSIZE 10//定义顺序表的最大容量
typedef struct{
    int data[MAXSIZE];//静态分配数组
    int length;//顺序表的当前长度

}sqlist;
void Initlist(sqlist *L){//初始化顺序表
    for(int i=0;i<MAXSIZE;i++){//遍历数组，将所有元素初始化为0
        L->data[i]=0;
    }
    L->length=0;//初始化长度为0
}
int main(){
    sqlist L;
    for(int i=0;i<MAXSIZE;i++){//遍历数组，打印每个元素
        printf("initbefore:data[%d]=%d\n",i,L.data[i]);
    }
    Initlist(&L);//初始化顺序表
    for(int i=0;i<MAXSIZE;i++){//遍历数组，打印每个元素
        printf("initafter:data[%d]=%d\n",i,L.data[i]);
    }
    return 0;
}
```
